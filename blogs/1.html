    # TEXT_02文本标签.html

    <!DOCTYPE html><!--HTML5的编码格式-->
<html lang="zh-CN"><!--语言使用中文-->
<head>
    <meta charset="UTF-8"><!--使用UTF-8格式进行解读-->
    <title>Document</title>
</head>
<body>
    <p>
        <cite>《光辉岁月》</cite>这个为作品的标题
    </p>
    <p>
        <dfn>我爸得了MVP</dfn>和<dfn>阿米诺思</dfn>都是特殊语义词
    </p>
    <p>
        商品原价：<del>199</del><!-- 删除标签 -->
        现在：<ins>99</ins><!-- 插入标签 -->
    </p>
    <p>
        水的化学方程是 H<sub>2</sub>O，
        <!-- 下标签 -->
        8的三次方是这么写的 8<sup>3</sup>
        <!-- 上标签 -->
    </p>
    <p>
        我们之后会学到其他的标签就像：<code>input</code>
        <!-- 标注代码的标签 -->
    </p>
    <p>
        手机提示你，<samp>支付宝到账100元</samp>
        <!-- 标识设备输出 -->
    </p>
    <p>
        保存的快捷键是：<kbd>ctrl+s</kbd>
        <!-- 键盘文本 -->
    </p>
    <p>
        <abbr title="英雄联盟">LOL</abbr>是一个游戏
        <!-- 缩写标识 -->
    </p>
    <p>
        阿米诺思，这句话反过来就是：<bdo dir="rtl"><!-- rtl就是从右往左 -->阿米诺思</bdo>
        <!-- 更改文本方向 -->
    </p>
    <p>
        定义变量<code>let <var>a</var> = 1 </code>
        <!-- VAR表明变量 -->
    </p>
    <p>
        <small>本网站的内容仅供参考</small>
        <!-- 版权标签 ---------------很少用-->
    </p>
    <p>
        我买过<b>掠影</b>皮肤，很好用。
        <!-- 摘要产品名字，关键词 ---------------很少用-->
    </p>
    <p>
        草帽法师说：“<i>阿米诺斯</i>”。
        <!-- 原本是指人物的话，现在的大多是用于字体图标 -->
    </p>
    <p>
        你把你好的英文单词写错了，不是<u>heelo</u>
        <!-- 用于标注写错的单词和语句 ---------------很少用-->
    </p>
    <p>
        神鹰黑手哥说过：<q>装逼，我让你飞起来</q>。
        <!-- 短引用，效果是加了一个双引号 ---------------很少用-->
    </p>
    <p>
        有一首歌的歌词是：
        <blockquote>鸡你太美，鸡你太美，鸡你太美，鸡你太美，鸡你太美，鸡你太美，鸡你太美</blockquote>
        <!-- 长引用，效果是换到下一行，开头空两格 块元素---------------很少用-->
    </p>
    <P>
        我们学校的地址是：<address>合肥市瑶海区</address>
        <!-- 地址信息，换到下一行 块元素 -->
    </P>
     <!-- 块元素独占一行 -->
</body>
</html>

    # TEXT_07表格.html

    <!DOCTYPE html>
<html lang="zh-CN">
<head>
    <meta charset="UTF-8">
    <title>Document</title>
</head>
<body>
    <table border="1" width="500" height="200" cellspacing="0">
        <!-- border是加边框，1是像素点的单位
         width是宽度
         height是高度
         cellspacing是单元框之间的距离
         -->
        <caption>学生信息</caption>
        <!-- 表的标题 不能加属性-->
        <thead align="center">
            <!-- align是水平对齐align='right'是右对齐，align='left'是左对齐align='center'是水平居中 
            valign是垂直对齐  valign='top'是上对齐   valign='middle'是中间对齐  valign='bottom'是底部对齐
            表的头部 能加height属性-->
            <tr>
                <th height="50" width="50" align="center" valign="middle">姓名</th>
                <th>性别</th>
            </tr>
        </thead>
        <tbody height="100" align="center" valign="middle">
            <tr height="50" align="center" valign="middle">
                <td height="50" width="50" align="center" valign="middle">王五</td>
                <td>男</td>
            </tr>
            <tr>
                <td>张三</td>
                <td>女</td>
            </tr>
        </tbody>
        <!-- 表的身体 -->
        <tfoot height="100" align="center" valign="middle">
            <tr>
                <td></td>
                <td></td>
            </tr>
        </tfoot>
        <!-- 表的脚注 -->
    </table>
</body>
</html>

    # TEXT_11表的_常用的控件.html

    <!DOCTYPE html>
<html lang="zh-CN">
<head>
    <meta charset="UTF-8">
    <title>Document</title>
</head>
<body>
    <form action="https://search.jd.com/search">
        <!-- 文本输入框 -->
        账户：<input type="text" name="account" value="zhangsan" maxlength="10"><br>
        <!-- value是指输入框默认值
         maxlength是指输入框的长度 -->
         <!-- 密码输入框 -->
        密码：<input type="password" name="pwd" value="123456" maxlength="10"><br>
        <!-- 单选框 -->
        性别：
        <input type="radio" name="gender" value="male">男
        <input type="radio" name="gender" value="female" checked>女<br>
        <!-- checked是默认勾选值 -->
         <!-- 多选框 -->
        爱好：
        <input type="checkbox" name="hobby" value="smoke">抽烟
        <input type="checkbox" name="hobby" value="drink">喝酒
        <input type="checkbox" name="hobby" value="perm">烫头<br>
        <!-- 文本域 -->
        其他：
        <textarea name="other" cols="22" rows="4"></textarea>
        <!-- cols和rows是调节文本域的列和行 -->
        籍贯：
        <select name="place">
            <option value="冀">河北</option>
            <option value="鲁">山东</option>
            <option value="皖" selected>安徽</option>
            <!-- selected默认选中 -->
        </select>
        <!-- 隐藏域 用户不可见的一个输入区域，作用是：提交表单的时候，携带一些固定的数据-->
        <input type="hidden" name="form" value="toutiao "><br>
        <!-- 确认按钮，第一种写法 -->
        <button type="submit">确认</button>
        <!-- 确认按钮，第二种写法 -->
        <!-- <input type="submit" value="确认"> -->
        <!-- 重置按钮_第一种写法 -->
        <button type="reset">重置</button>
        <!-- 重置按钮_第二种写法 -->
        <!-- <input type="reset" value="点我重置"> -->
        <!-- 普通按钮 -->
        <button type="button">检查账号是否被注册</button>
        <!-- <input type="button" value="检查账号是否被注册"> -->
    </form>
</body>
</html>

    # TEXT_15meta元信息.html

    <!DOCTYPE html>
<html lang="zh-CN">
<head>
    <!--配置字符编码--> 
    <meta charset="UTF-8">
    <!-- 针对IE浏览器的一个兼容性设置 -->
    <meta http-equiv="X-UA-Compatible" content="IE=edge">
    <!-- 针对移动端的一个配置 -->
    <meta name="viewport" content="width=device-width,initial-scale=1.0">
    <!-- 配置网页的关键字 -->
    <meta name="keyword" content="网上购物，游戏下载">
    <!-- 配置网页描述信息 -->
    <meta name="description" content="成立于2000年的公司">
    <!-- 针对搜索引擎爬虫配置 -->
    <meta name="robots" content="index">
    <!-- 配置网页作者 -->
    <meta name="author" content="tony">
    <!-- 配置网页生成工具 -->
    <meta name="generator" content="Visual Studio Code">
    <!-- 配置定义网页版权信息 -->
    <meta name="copyright" content="2023-2027&copy;版权所有">
    <!-- 配置定义网页自动刷新 10秒钟自动刷新到其他网站-->
    <meta http-equiv="refresh" content="10;url=http://www.baidu.com"> 
    <!-- 三秒后原地刷新 -->
    <meta http-equiv="refresh" charset="3">
    <title>meta元信息</title>
</head>
<body>

</body>
</html>
